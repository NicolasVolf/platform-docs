    Meus bottlenecks

    O curso pede ao menos 2 bottlenecks por integrante. Implementei dois, ambos da tabela de bottlenecks do curso. Cada um tem a página completa (problema, solução, código, evidência) na documentação do grupo; aqui está o resumo com os números que mostram o efeito.

    Bottleneck Número que mostra o efeito Página completa

    Autorização (RBAC) as 4 brechas (listar, ver, apagar contas alheias e usar a rota interna) passaram a dar 403; 10/10 testes manuais e 13 testes automatizados Autorização (RBAC)

    Mensageria (Kafka)

    Problema: depois do cadastro, a loja manda um e-mail de boas-vindas. Se o account chamasse o serviço de e-mail dentro do POST /accounts, o cadastro ficaria acoplado a ele: o usuário esperaria a latência do provedor, uma queda do e-mail viraria uma queda do cadastro, e um pico de cadastros bateria no provedor no mesmo ritmo.

    Solução: segue o Hands-on de mensageria do curso (Kafka, transactional outbox, consumidor idempotente e DLT), adaptado para o cadastro de contas.

    flowchart LR
    u([Usuário]) -->|"POST /auth/register"| auth --> account
    subgraph tx ["uma transação no PostgreSQL"]
        account -->|"INSERT"| acc[("accounts")]
        account -->|"INSERT AccountCreated"| ob[("outbox")]
    end
    relay["OutboxRelay<br/>(a cada 200 ms)"] -->|"lê pendentes"| ob
    relay -->|"chave = id da conta<br/>acks=all"| k[["Kafka<br/>account-events (6 partições)"]]
    k --> n["notification-service"]
    n -->|"dedupe por eventId"| pe[("processed_events")]
    n -->|SMTP| mail(["Mailpit"])
    n -. "falha permanente" .-> dlt[["account-events-dlt"]]
    Decisão Motivo
    Transactional outbox gravar a conta e publicar no Kafka são duas escritas em sistemas diferentes; se o processo cair entre elas, o evento se perde. Com o outbox, conta e evento entram na mesma transação e um relay publica depois do commit
    Relay com FOR UPDATE SKIP LOCKED várias réplicas do account podem rodar o relay sem publicar a mesma linha duas vezes
    Produtor com acks=all e idempotência o broker só confirma depois de gravar, e um reenvio não duplica a mensagem
    Chave = id da conta, 6 partições eventos da mesma conta ficam em ordem (mesma partição); 6 é o teto de consumidores em paralelo
    Consumidor idempotente a entrega é at-least-once: o eventId é gravado em processed_events na mesma transação do e-mail; se já existe, é ignorado
    3 retries com backoff e DLT falha temporária é tentada de novo; depois disso a mensagem vai para a DLT e a partição continua andando. Mensagem inválida vai direto para a DLT
    O evento não leva senha nem hash só o mínimo para o e-mail

    package store.account.outbox;

import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;
import java.util.UUID;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.TimeUnit;

import org.apache.kafka.clients.producer.ProducerRecord;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.kafka.support.SendResult;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Transactional;

/** Publica no Kafka os eventos que o Outbox gravou, depois do commit (Hands-on 7). */
@Component
public class OutboxRelay {

    private static final Logger log = LoggerFactory.getLogger(OutboxRelay.class);

    record Pending(UUID id, String aggregateId, String eventType, String payload) {
    }

    private final JdbcClient jdbc;
    private final JdbcTemplate jdbcTemplate;
    private final KafkaTemplate<String, String> kafka;
    private final String topic;

    public OutboxRelay(JdbcClient jdbc, JdbcTemplate jdbcTemplate,
                       KafkaTemplate<String, String> kafka,
                       @Value("${store.account.topic}") String topic) {
        this.jdbc = jdbc;
        this.jdbcTemplate = jdbcTemplate;
        this.kafka = kafka;
        this.topic = topic;
    }

    @Scheduled(fixedDelayString = "${store.account.relay-interval-ms:200}")
    @Transactional
    public void relay() throws Exception {
        // SKIP LOCKED: várias réplicas do account podem rodar o relay sem pegar as mesmas linhas
        List<Pending> batch = jdbc.sql("""
                SELECT id, aggregate_id, event_type, payload::text AS payload
                FROM accounts.outbox
                WHERE published_at IS NULL
                ORDER BY created_at
                LIMIT 100
                FOR UPDATE SKIP LOCKED
                """)
                .query(Pending.class)
                .list();
        if (batch.isEmpty()) {
            return;
        }

        List<CompletableFuture<SendResult<String, String>>> acks = new ArrayList<>();
        for (Pending p : batch) {
            // chave = id da conta: eventos da mesma conta caem na mesma partição (ordem por chave)
            ProducerRecord<String, String> record =
                    new ProducerRecord<>(topic, p.aggregateId(), p.payload());
            record.headers().add("eventId", p.id().toString().getBytes(StandardCharsets.UTF_8));
            record.headers().add("eventType", p.eventType().getBytes(StandardCharsets.UTF_8));
            acks.add(kafka.send(record));
        }
        // espera o ack de todos; se falhar, a transação volta e as linhas são tentadas de novo
        CompletableFuture.allOf(acks.toArray(new CompletableFuture<?>[0])).get(10, TimeUnit.SECONDS);

        jdbcTemplate.batchUpdate("UPDATE accounts.outbox SET published_at = clock_timestamp() WHERE id = ?",
                batch.stream().map(p -> new Object[] {p.id()}).toList());
        log.info("relayed {} event(s) to {}", batch.size(), topic);
    }

    @Scheduled(cron = "0 0 * * * *")
    @Transactional
    public void purge() {
        int n = jdbcTemplate.update(
                "DELETE FROM accounts.outbox WHERE published_at < now() - interval '1 day'");
        log.info("purged {} published outbox row(s)", n);
    }
}

    package store.notification;

import java.time.OffsetDateTime;
import java.time.ZoneOffset;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.jdbc.core.simple.JdbcClient;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class NotificationService {

    private static final Logger log = LoggerFactory.getLogger(NotificationService.class);

    private final JdbcClient jdbc;
    private final EmailSender email;
    private final NotificationProperties props;

    public NotificationService(JdbcClient jdbc, EmailSender email, NotificationProperties props) {
        this.jdbc = jdbc;
        this.email = email;
        this.props = props;
    }

    /**
     * Entrega at-least-once: o mesmo evento pode chegar duas vezes. O eventId é
     * gravado na mesma transação do registro do e-mail; se já existir, é duplicata.
     *
     * @return true se enviou o e-mail, false se o evento era duplicado
     */
    @Transactional
    public boolean handle(AccountEvent e) {
        if (props.idempotent()) {
            int fresh = jdbc.sql("INSERT INTO processed_events (event_id) VALUES (?) ON CONFLICT DO NOTHING")
                    .param(e.eventId())
                    .update();
            if (fresh == 0) {
                log.info("duplicate event {} for account {} ignored", e.eventId(), e.accountId());
                return false;
            }
        }
        email.welcome(e); // se lançar, a transação inteira volta (inclusive a linha de deduplicação)
        jdbc.sql("""
                INSERT INTO notifications (event_id, account_id, email, kind, occurred_at)
                VALUES (?, ?, ?, ?, ?)
                """)
                .params(e.eventId(), e.accountId(), e.email(), e.eventType(),
                        OffsetDateTime.ofInstant(e.occurredAt(), ZoneOffset.UTC))
                .update();
        log.info("welcome e-mail sent for account {}", e.accountId());
        return true;
    }
}

    package store.notification;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.kafka.core.KafkaTemplate;
import org.springframework.kafka.listener.DeadLetterPublishingRecoverer;
import org.springframework.kafka.listener.DefaultErrorHandler;
import org.springframework.kafka.support.ExponentialBackOffWithMaxRetries;

import tools.jackson.core.JacksonException;

@Configuration
public class KafkaConfig {

    /** Usado pela fábrica de listeners do Spring Boot (um único bean CommonErrorHandler). */
    @Bean
    DefaultErrorHandler errorHandler(KafkaTemplate<String, String> template) {
        // depois das tentativas, publica em <tópico>-dlt, mesma partição, com a exceção nos headers
        DeadLetterPublishingRecoverer recoverer = new DeadLetterPublishingRecoverer(template);

        ExponentialBackOffWithMaxRetries backOff = new ExponentialBackOffWithMaxRetries(3);
        backOff.setInitialInterval(500L);  // 0,5 s, 1 s, 2 s
        backOff.setMultiplier(2.0);
        backOff.setMaxInterval(5_000L);

        DefaultErrorHandler handler = new DefaultErrorHandler(recoverer, backOff);
        // falhas permanentes: tentar de novo não adianta, vai direto para a DLT
        handler.addNotRetryableExceptions(JacksonException.class, IllegalArgumentException.class);
        return handler;
    }
}

    Autorização (RBAC)

    Problema: o gateway já garantia quem é o usuário, mas não o que ele podia fazer. Achei a falha lendo o gateway e o account: qualquer pessoa logada, com o próprio token, conseguia pelo gateway listar todas as contas, ver e apagar a conta de outra pessoa e usar a rota interna POST /accounts/login. É o gargalo Authentication & Authorization da tabela do curso, e a mitigação sugerida é RBAC.

    Solução: dois papéis, USER e ADMIN. O papel mora no Account, viaja dentro do JWT assinado e é conferido no gateway, o único ponto de entrada:

    Rota escolhida pelo gateway USER ADMIN
    account (/accounts/**) só GET /accounts/{o próprio id} tudo
    auth, exchange e as demais tudo (basta estar autenticado) tudo

    Sem token ou com token inválido continua sendo 401; token válido sem permissão é 403.

    Decisão Motivo
    Papel no JWT, conferido no gateway o gateway já resolve o token em toda requisição (com cache): a checagem não acrescenta chamada de rede
    A regra olha a rota escolhida, não o texto da URL variações como /accounts;x=1 ou /accounts/ caem na mesma regra
    Negado por padrão no account qualquer rota nova do account exige ADMIN sem precisar lembrar de atualizar a regra
    Conta nova é sempre USER; ADMIN só direto no banco ninguém se autopromove: um "role": "ADMIN" no cadastro é ignorado (testado)
    Token antigo sem o claim role vale USER na dúvida, o menor acesso

    package store.gateway.security;

import java.util.List;

import org.springframework.cloud.gateway.route.Route;
import org.springframework.cloud.gateway.support.ServerWebExchangeUtils;
import org.springframework.http.HttpMethod;
import org.springframework.http.server.PathContainer;
import org.springframework.stereotype.Component;
import org.springframework.web.server.ServerWebExchange;

import io.micrometer.core.instrument.Counter;
import io.micrometer.core.instrument.MeterRegistry;

/**
 * Bottleneck: autenticação e autorização (RBAC).
 *
 * O AuthorizationFilter já garantiu QUEM é o usuário (token válido); aqui se
 * decide O QUE ele pode fazer, pelo papel que veio no JWT:
 *
 * | rota escolhida | USER                             | ADMIN |
 * |----------------|----------------------------------|-------|
 * | account        | só GET /accounts/{o próprio id}  | tudo  |
 * | demais (auth, exchange, ...) | tudo (basta estar autenticado) | tudo |
 *
 * Decisões:
 * - a regra olha a ROTA que o gateway escolheu, não o texto da URL: variações
 *   como "/accounts;x=1" ou "//accounts" caem na mesma rota e na mesma regra;
 * - no account vale "negado por padrão": qualquer coisa fora da única exceção
 *   do USER (inclusive rotas novas que o account ganhar) exige ADMIN;
 * - o id é comparado já decodificado e sem parâmetros de segmento.
 */
@Component
public class AccessPolicy {

    static final String ACCOUNT_ROUTE = "account";

    private final Counter allowed;
    private final Counter denied;

    public AccessPolicy(MeterRegistry registry) {
        this.allowed = counter(registry, "allow");
        this.denied = counter(registry, "deny");
    }

    private static Counter counter(MeterRegistry registry, String result) {
        return Counter.builder("gateway.rbac.decision")
            .description("Decisoes do RBAC nas rotas autenticadas (allow, deny)")
            .tag("result", result)
            .register(registry);
    }

    public boolean allows(ServerWebExchange exchange, Identity who) {
        final boolean ok = decide(exchange, who);
        (ok ? allowed : denied).increment();
        return ok;
    }

    private static boolean decide(ServerWebExchange exchange, Identity who) {
        if (who.isAdmin()) return true;
        final Route route = exchange.getAttribute(ServerWebExchangeUtils.GATEWAY_ROUTE_ATTR);
        if (route == null || !ACCOUNT_ROUTE.equals(route.getId())) return true;
        return HttpMethod.GET.equals(exchange.getRequest().getMethod())
            && isOwnAccount(exchange.getRequest().getPath().pathWithinApplication(), who.idAccount());
    }

    /** true só para exatamente /accounts/{idAccount}. */
    static boolean isOwnAccount(PathContainer path, String idAccount) {
        final List<PathContainer.PathSegment> segments = path.elements().stream()
            .filter(PathContainer.PathSegment.class::isInstance)
            .map(PathContainer.PathSegment.class::cast)
            .filter(s -> !s.value().isEmpty())
            .toList();
        return segments.size() == 2
            && segments.stream().allMatch(s -> s.parameters().isEmpty())
            && "accounts".equals(segments.get(0).valueToMatch())
            && idAccount != null
            && idAccount.equals(segments.get(1).valueToMatch());
    }

}
